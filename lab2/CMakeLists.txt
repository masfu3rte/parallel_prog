cmake_minimum_required(VERSION 3.16)
project(parallel_lab_2 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(lab2 src/main.cpp)
target_compile_features(lab2 PRIVATE cxx_std_17)
set_target_properties(lab2 PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(lab2 PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(lab2 PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(lab2 PRIVATE -Wall -Wextra -Wpedantic)
endif()
